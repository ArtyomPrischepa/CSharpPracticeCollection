using System;
using System.Linq;

namespace LinqAggregatePractice
{
    // Модель данных для работы
    public record class Product(string Name, decimal Price, int StockCount);

    class Program
    {
        static void Main(string[] args)
        {
            // Исходные данные: Магазин электроники
            Product[] shopInventory = {
                new Product("Смартфон", 45000m, 12),
                new Product("Ноутбук", 89900m, 5),
                new Product("Наушники", 5500m, 24),
                new Product("Умные часы", 15000m, 0), // Нет на складе
                new Product("Монитор", 23400m, 8),
                new Product("Клавиатура", 3200m, 15)
            };

            Console.WriteLine("=== ПРАКТИЧЕСКИЙ ТЕСТ: LINQ AGGREGATE ===\n");

            // ----------------------------------------------------------------------------
            // ЗАДАНИЕ 1: Использование Count()
            // Посчитайте, сколько ВСЕГО видов товаров есть на складе (StockCount > 0)
            // ----------------------------------------------------------------------------
            int availableProductsCount = shopInventory.Count(x=>x.StockCount>0);
            Console.WriteLine($"1. Видов товаров в наличии: {availableProductsCount} (Ожидается: 5)");


            // ----------------------------------------------------------------------------
            // ЗАДАНИЕ 2: Использование Max() и Min()
            // Найдите самую высокую и самую низкую цену товара в магазине
            // ----------------------------------------------------------------------------
            decimal maxPrice = shopInventory.Max(x=>x.Price);
            decimal minPrice = shopInventory.Min(x => x.Price);
            Console.WriteLine($"2. Максимальная цена: {maxPrice} руб. | Минимальная: {minPrice} руб.");


            // ----------------------------------------------------------------------------
            // ЗАДАНИЕ 3: Использование Average()
            // Найдите среднюю стоимость товара в каталоге
            // ----------------------------------------------------------------------------
            decimal averagePrice = shopInventory.Average(x=>x.Price);
            Console.WriteLine($"3. Средняя цена товара: {averagePrice:F2} руб. (Ожидается: 30333,33)");


            // ----------------------------------------------------------------------------
            // ЗАДАНИЕ 4: Использование Sum()
            // Посчитайте общую стоимость всей продукции на складе. 
            // Подсказка: для каждого товара нужно умножить его Price на StockCount, а затем просуммировать.
            // ----------------------------------------------------------------------------
            decimal totalInventoryValue = shopInventory.Sum(x=>x.Price * x.StockCount);
            Console.WriteLine($"4. Общая стоимость всех товаров на складе: {totalInventoryValue} руб. (Ожидается: 1326400)");


            // ----------------------------------------------------------------------------
            // ЗАДАНИЕ 5: Использование Aggregate()
            // Сделайте красивую текстовую строку со списком всех названий товаров через запятую.
            // Начальное значение (seed) задайте как "Каталог:". 
            // Результат должен быть в формате: "Каталог: Смартфон, Ноутбук, Наушники, ..."
            // ----------------------------------------------------------------------------
            string productCatalogString = shopInventory.Aggregate(
                "Каталог: ",
                (current, nextProduct) =>  current  + nextProduct.Name + ", "
            );
            Console.WriteLine($"5. {productCatalogString}");

            Console.ReadLine();
        }
    }
}
