using System;
using System.Linq;
using System.Threading;
using System.Threading.Tasks;

class Program
{
    static void Main()
    {
        // Данные: массив строк, но одна строка сломана (содержит буквы вместо суммы)
        string[] transactions = { "100", "200", "ОШИБКА", "400", "500" };

        // Инициализируйте CancellationTokenSource
        // НАПИШИТЕ КОД ЗДЕСЬ (создайте cts)
        var cts = new CancellationTokenSource();
        // Запуск фоновой задачи для имитации отмены операции через 500 миллисекунд
        Task.Run(() =>
        {
            Thread.Sleep(500);
            cts.Cancel();
            // Вызовите отмену через ваш cts
            // НАПИШИТЕ КОД ЗДЕСЬ (вызов метода отмены)
        });

        try
        {
            // Постройте PLINQ-запрос:
            // 1. Сделайте транзакции параллельными (.AsParallel)
            // 2. Примените токен отмены (.WithCancellation)
            // 3. Спроецируйте строку в число с помощью метода ParseTransaction (.Select)
            var processed = transactions.AsParallel().WithCancellation(cts.Token).Select(x => ParseTransaction(x));// НАПИШИТЕ КОД ЗДЕСЬ

            // Запустите выполнение и вывод результатов
            foreach (var amount in processed)
            {
                Console.WriteLine($"Успешно обработано: {amount} руб.");
            }
        }
        // Перехватите исключение отмены операции
        catch (/* НАПИШИТЕ ТИП ИСКЛЮЧЕНИЯ ИМЕННО ДЛЯ ОТМЕНЫ */OperationCanceledException)
        {
            Console.WriteLine("Операция была принудительно прервана пользователем или таймаутом.");
        }
        // Перехватите агрегированное исключение (ошибку парсинга "ОШИБКА")
        catch (/* НАПИШИТЕ ТИП АГРЕГИРОВАННОГО ИСКЛЮЧЕНИЯ */AggregateException ex)
        {
            Console.WriteLine("Произошли ошибки при обработке данных:");
            foreach( var iex in ex.InnerExceptions)
            {
                Console.WriteLine(iex.Message);
            }// Переберите все внутренние исключения в ex.InnerExceptions и выведите их e.Message
            // НАПИШИТЕ ЦИКЛ FOREACH ЗДЕСЬ
        }
        finally
        {
            cts?.Dispose();// Очистите ресурсы токена (вызовите Dispose)
            // НАПИШИТЕ КОД ЗДЕСЬ
        }
    }

    // Метод-помощник, имитирующий долгую обработку и парсинг
    static int ParseTransaction(string input)
    {
        Thread.Sleep(300); // Имитация задержки сети
        return int.Parse(input); // Вызовет FormatException на строке "ОШИБКА"
    }
}
