using System;
using System.Collections.Generic;
using System.Threading;
using System.Threading.Tasks;

class Program
{
    static void Main()
    {
        Console.WriteLine("--- Старт программы ---\n");

        // ЗАДАЧА 1: Запустите параллельно три действия с помощью Parallel.Invoke.
        // Первое — выводит "Действие 1", второе — "Действие 2", третье — задержка Thread.Sleep(1000).

        // Напишите Parallel.Invoke здесь:
        Parallel.Invoke(() => Console.WriteLine("asd"), () => Console.WriteLine("asd2"), () => Console.WriteLine("asd3"));


        Console.WriteLine("\n-----------------------\n");


        // ЗАДАЧА 2: Переберите список имен параллельно с помощью Parallel.ForEach.
        // Если длина имени равна 4 символам ("Ivan"), прервите цикл с помощью ParallelLoopState.
        // Выведите в консоль результат: завершился ли цикл полностью (IsCompleted).
        var users = new List<string> { "Tom", "Bob", "Alice", "Ivan", "Sam" };

        // Напишите Parallel.ForEach здесь:
        ParallelLoopResult result = Parallel.ForEach(users,(x, pls) => {
            if (x.Length == 4)
                pls.Break();
                Console.WriteLine(x);
        });

        Console.WriteLine($"Цикл завершен полностью: {result.IsCompleted}");

        Console.WriteLine("\n--- Конец программы ---");
        Console.ReadLine();
    }
}
